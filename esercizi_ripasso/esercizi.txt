def log_calls(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        # args è una tupla contenente tutti gli argomenti posizionali
        print(f"Chiamata a: {func.__name__}")
        print(f"Argomenti (tupla): {args}")
        print(f"Keyword args (dict): {kwargs}")
        
        return func(*args, **kwargs)
    return wrapper

# Applicazione del decoratore
@log_calls
def calcola_totale(a, b, sconto=0):
    return (a + b) - sconto

# Test della funzione
risultato = calcola_totale(100, 50, sconto=10)
print(f"Risultato finale: {risultato}")

async def task_a():
    # Simula un'operazione di I/O (come il download di un file)
    await asyncio.sleep(2)
    return "Task A completato!"

async def main():
    # L'espressione 'await' sospende l'esecuzione di main(), 
    # restituendo il controllo al ciclo degli eventi (event loop).
    # Il codice riprende solo quando task_a() ha terminato il suo lavoro.
    risultato = await task_a()
    print(risultato)

# Avvio del programma
if __name__ == "__main__":
    asyncio.run(main())
